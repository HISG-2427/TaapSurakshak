    <button type="button" onclick="sendMessage()">Send WhatsApp Message</button>


    <div id="status" class="status"></div>


    <script>

        async function generateAISuggestion() {

            const prompt = document.getElementById('prompt').value;

            const statusDiv = document.getElementById('status');

            

            if (!prompt) return alert('Please input an AI prompt.');

            statusDiv.textContent = 'Generating suggestion...';


            try {

                const response = await fetch('/api/generate-suggestion', {

                    method: 'POST',

                    headers: { 'Content-Type': 'application/json' },

                    body: JSON.stringify({ prompt })

                });

                const data = await response.json();


                if (response.ok) {

                    document.getElementById('message').value = data.suggestion;

                    statusDiv.textContent = 'Suggestion ready!';

                } else {

                    statusDiv.textContent = 'Error: ' + data.error;

                }

            } catch (err) {

                statusDiv.textContent = 'Failed to retrieve AI response.';

            }

        }


        async function sendMessage() {

            const number = document.getElementById('number').value;

            const message = document.getElementById('message').value;

            const statusDiv = document.getElementById('status');


            if (!number || !message) return alert('Both Phone Number and Message are required.');

            statusDiv.textContent = 'Sending message...';


            try {

                const response = await fetch('/api/send-message', {

                    method: 'POST',

                    headers: { 'Content-Type': 'application/json' },

                    body: JSON.stringify({ number, message })

                });

                const data = await response.json();


                if (data.success) {

                    statusDiv.textContent = '✅ Message sent successfully!';

                } else {

                    statusDiv.textContent = '❌ Error: ' + data.error;

                }

            } catch (err) {

                statusDiv.textContent = 'Failed to submit message request.';

            }

        }
